cmake_minimum_required(VERSION 3.16)
project(QtGLTriangle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
add_executable(QtGLTriangle main.cpp)
target_link_libraries(QtGLTriangle PRIVATE Qt${QT_VERSION_MAJOR}::Widgets)
if(QT_VERSION_MAJOR EQUAL 6)
    find_package(Qt6 REQUIRED COMPONENTS OpenGLWidgets)
    target_link_libraries(QtGLTriangle PRIVATE Qt6::OpenGLWidgets)
endif()
