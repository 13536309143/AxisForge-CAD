cmake_minimum_required(VERSION 3.16)
project(AxisForgeCAD VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
if(QT_VERSION_MAJOR EQUAL 6)
  find_package(Qt6 REQUIRED COMPONENTS OpenGLWidgets)
endif()
add_library(axisforge_core src/core/Scene.cpp src/core/Geometry.cpp src/core/Constraints.cpp src/core/ProjectIO.cpp)
target_include_directories(axisforge_core PUBLIC src)
target_link_libraries(axisforge_core PUBLIC Qt${QT_VERSION_MAJOR}::Gui)
add_executable(AxisForgeCAD WIN32 src/main.cpp src/app/Document.cpp src/ui/MainWindow.cpp src/render/Viewport.cpp)
target_link_libraries(AxisForgeCAD PRIVATE axisforge_core Qt${QT_VERSION_MAJOR}::Widgets)
if(QT_VERSION_MAJOR EQUAL 6)
  target_link_libraries(AxisForgeCAD PRIVATE Qt6::OpenGLWidgets)
endif()
if(MSVC)
  target_compile_options(axisforge_core PRIVATE /utf-8 /W4)
  target_compile_options(AxisForgeCAD PRIVATE /utf-8 /W4)
else()
  target_compile_options(axisforge_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(AxisForgeCAD PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(axisforge_tests tests/core_tests.cpp)
  target_link_libraries(axisforge_tests PRIVATE axisforge_core)
  if(MSVC)
    target_compile_options(axisforge_tests PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME core COMMAND axisforge_tests)
  add_test(NAME desktop_smoke COMMAND AxisForgeCAD --self-test ${CMAKE_CURRENT_BINARY_DIR}/test-artifacts)
  set_tests_properties(desktop_smoke PROPERTIES TIMEOUT 45)
endif()
install(TARGETS AxisForgeCAD RUNTIME DESTINATION bin)
